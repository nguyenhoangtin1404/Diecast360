### Test Specs: `tests/e2e/auth-security.spec.ts`

```typescript
// Spec 1: X-Trace-Id header
test('POST /auth/login returns X-Trace-Id header', async ({ request }) => {
  const res = await request.post('/api/v1/auth/login', { data: { email, password } });
  expect(res.headers()['x-trace-id']).toBeTruthy();
  // verify UUIDv7 format
});

// Spec 2: Account lockout after N failed attempts
test('Account locked after max failed attempts', async ({ request }) => {
  for (let i = 0; i < 3; i++) {
    await request.post('/api/v1/auth/login', { data: { email, password: 'wrong' } });
  }
  const res = await request.post('/api/v1/auth/login', { data: { email, password: 'wrong' } });
  expect(res.status()).toBe(429);
  const body = await res.json();
  expect(body.error).toBe('AUTH_ACCOUNT_LOCKED');
  expect(body.lockedUntil).toBeTruthy();
});

// Spec 3: Admin can unlock account
test('Admin unlock resets lockout', async ({ request }) => {
  // ... lock account first
  const unlock = await request.post('/api/v1/admin/auth/unlock',
    { data: { email }, headers: { Cookie: adminCookie } });
  expect(unlock.status()).toBe(204);
  // login should succeed now
});

// Spec 4: Login success resets counter
test('Successful login resets failed attempt counter', async ({ request }) => {
  // 2 failed attempts
  // 1 success
  // 2 more failed attempts — should NOT be locked (counter reset)
});

// Spec 5: Rate limit response format

});
```
